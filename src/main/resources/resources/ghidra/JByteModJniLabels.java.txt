//@category JByteMod
import ghidra.app.script.GhidraScript;
import ghidra.program.model.address.Address;
import ghidra.program.model.data.BooleanDataType;
import ghidra.program.model.data.DataType;
import ghidra.program.model.data.DoubleDataType;
import ghidra.program.model.data.FloatDataType;
import ghidra.program.model.data.IntegerDataType;
import ghidra.program.model.data.LongLongDataType;
import ghidra.program.model.data.PointerDataType;
import ghidra.program.model.data.ShortDataType;
import ghidra.program.model.data.SignedByteDataType;
import ghidra.program.model.data.UnsignedShortDataType;
import ghidra.program.model.data.VoidDataType;
import ghidra.program.model.listing.Function;
import ghidra.program.model.listing.ParameterImpl;
import ghidra.program.model.listing.ReturnParameterImpl;
import ghidra.program.model.listing.Variable;
import ghidra.program.model.mem.Memory;
import ghidra.program.model.mem.MemoryBlock;
import ghidra.program.model.symbol.SourceType;
import ghidra.program.model.symbol.Symbol;
import ghidra.program.model.symbol.SymbolIterator;

import java.io.BufferedReader;
import java.io.File;
import java.nio.ByteBuffer;
import java.nio.ByteOrder;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class JByteModJniLabels extends GhidraScript {
    private static final String DATA_FILE = "jbytemod_natives.tsv";
    private static final long MAX_BLOCK = 256L * 1024 * 1024;
    private final Set<Address> annotated = new HashSet<>();

    private record Row(String type, String symbol, String className, String method, String descriptor, boolean isStatic) {
    }

    @Override
    protected void run() throws Exception {
        File scriptDirectory = getSourceFile().getParentFile().getFile(false);
        File data = new File(scriptDirectory, DATA_FILE);
        if (!data.isFile()) {
            printerr("JByteMod: " + DATA_FILE + " was not found next to this script.");
            return;
        }
        String programName = currentProgram.getName();
        List<Row> exports = new ArrayList<>();
        List<Row> registered = new ArrayList<>();
        try (BufferedReader reader = Files.newBufferedReader(data.toPath(), StandardCharsets.UTF_8)) {
            String line;
            while ((line = reader.readLine()) != null) {
                String[] parts = line.split("\t", -1);
                if (parts.length < 7 || !parts[1].equals(programName)) {
                    continue;
                }
                Row row = new Row(parts[0], parts[2], parts[3], parts[4], parts[5], parts[6].equals("static"));
                if (row.type().equals("EXPORT")) {
                    exports.add(row);
                } else if (row.type().equals("REGISTER")) {
                    registered.add(row);
                }
            }
        }
        if (exports.isEmpty() && registered.isEmpty()) {
            println("JByteMod: nothing to apply for " + programName);
            return;
        }
        int applied = 0;
        for (Row row : exports) {
            Address address = findExport(row.symbol());
            if (address == null) {
                println("JByteMod: symbol not found: " + row.symbol());
                continue;
            }
            if (annotate(address, row, "JNI export " + row.symbol())) {
                applied++;
            }
        }
        println("JByteMod: annotated " + applied + " of " + exports.size() + " exported JNI function(s)");
        if (!registered.isEmpty()) {
            int resolved = resolveRegistered(registered);
            println("JByteMod: resolved " + resolved + " of " + registered.size() + " RegisterNatives candidate(s)");
        }
    }

    private Address findExport(String name) {
        SymbolIterator iterator = currentProgram.getSymbolTable().getSymbols(name);
        while (iterator.hasNext()) {
            Symbol symbol = iterator.next();
            if (symbol.getAddress().isMemoryAddress()) {
                return symbol.getAddress();
            }
        }
        return null;
    }

    private boolean annotate(Address address, Row row, String origin) throws Exception {
        Function function = getFunctionAt(address);
        if (function == null) {
            disassemble(address);
            function = createFunction(address, null);
        }
        List<String> arguments = parseArguments(row.descriptor());
        String signature = row.className() + "." + row.method() + "(" + String.join(", ", arguments) + ") : "
                + parseReturn(row.descriptor());
        boolean first = annotated.add(address);
        if (first) {
            setPlateComment(address, "Java: " + signature + "\n" + origin + "\nDescriptor: " + row.descriptor());
        } else {
            setPlateComment(address, getPlateComment(address) + "\nAlso: " + signature + " (" + row.descriptor() + ")");
        }
        createBookmark(address, "JByteMod", signature);
        if (function != null && first) {
            applySignature(function, row);
        }
        return true;
    }

    private void applySignature(Function function, Row row) {
        try {
            List<Variable> parameters = new ArrayList<>();
            parameters.add(new ParameterImpl("env", PointerDataType.dataType, currentProgram));
            parameters.add(new ParameterImpl(row.isStatic() ? "clazz" : "thiz", PointerDataType.dataType, currentProgram));
            List<String> descriptors = descriptorTypes(row.descriptor());
            for (int index = 0; index < descriptors.size(); index++) {
                parameters.add(new ParameterImpl("arg" + index, dataTypeFor(descriptors.get(index)), currentProgram));
            }
            String returnDescriptor = row.descriptor().substring(row.descriptor().indexOf(')') + 1);
            function.updateFunction(null, new ReturnParameterImpl(dataTypeFor(returnDescriptor), currentProgram),
                    parameters, Function.FunctionUpdateType.DYNAMIC_STORAGE_FORMAL_PARAMS, true, SourceType.USER_DEFINED);
        } catch (Exception failure) {
            printerr("JByteMod: could not set the signature of " + function.getName() + ": " + failure.getMessage());
        }
    }

    private static DataType dataTypeFor(String descriptor) {
        return switch (descriptor.charAt(0)) {
            case 'Z' -> BooleanDataType.dataType;
            case 'B' -> SignedByteDataType.dataType;
            case 'C' -> UnsignedShortDataType.dataType;
            case 'S' -> ShortDataType.dataType;
            case 'I' -> IntegerDataType.dataType;
            case 'J' -> LongLongDataType.dataType;
            case 'F' -> FloatDataType.dataType;
            case 'D' -> DoubleDataType.dataType;
            case 'V' -> VoidDataType.dataType;
            default -> PointerDataType.dataType;
        };
    }

    private static List<String> descriptorTypes(String descriptor) {
        List<String> result = new ArrayList<>();
        int index = descriptor.indexOf('(') + 1;
        int end = descriptor.indexOf(')');
        while (index > 0 && index < end) {
            int start = index;
            while (descriptor.charAt(index) == '[') {
                index++;
            }
            if (descriptor.charAt(index) == 'L') {
                index = descriptor.indexOf(';', index) + 1;
            } else {
                index++;
            }
            result.add(descriptor.substring(start, index));
        }
        return result;
    }

    private int resolveRegistered(List<Row> rows) throws Exception {
        Memory memory = currentProgram.getMemory();
        int pointerSize = currentProgram.getDefaultPointerSize();
        ByteOrder order = currentProgram.getLanguage().isBigEndian() ? ByteOrder.BIG_ENDIAN : ByteOrder.LITTLE_ENDIAN;

        List<MemoryBlock> blocks = new ArrayList<>();
        List<byte[]> contents = new ArrayList<>();
        for (MemoryBlock block : memory.getBlocks()) {
            if (!block.isInitialized() || block.getSize() > MAX_BLOCK || block.getSize() < pointerSize) {
                continue;
            }
            byte[] bytes = new byte[(int) block.getSize()];
            try {
                block.getBytes(block.getStart(), bytes);
            } catch (Exception failure) {
                continue;
            }
            blocks.add(block);
            contents.add(bytes);
        }

        Map<String, Set<Long>> stringAddresses = new HashMap<>();
        Set<String> needles = new HashSet<>();
        for (Row row : rows) {
            needles.add(row.method());
            needles.add(row.descriptor());
        }
        for (String needle : needles) {
            byte[] pattern = (needle + "\0").getBytes(StandardCharsets.UTF_8);
            Set<Long> addresses = new HashSet<>();
            for (int index = 0; index < blocks.size(); index++) {
                byte[] bytes = contents.get(index);
                long base = blocks.get(index).getStart().getOffset();
                for (int position = indexOf(bytes, pattern, 0); position >= 0; position = indexOf(bytes, pattern, position + 1)) {
                    addresses.add(base + position);
                }
            }
            stringAddresses.put(needle, addresses);
        }

        Map<Long, List<Row>> byNameAddress = new HashMap<>();
        for (Row row : rows) {
            for (long address : stringAddresses.getOrDefault(row.method(), Set.of())) {
                byNameAddress.computeIfAbsent(address, key -> new ArrayList<>()).add(row);
            }
        }

        Set<Row> resolved = new HashSet<>();
        for (int index = 0; index < blocks.size(); index++) {
            MemoryBlock block = blocks.get(index);
            if (block.isExecute()) {
                continue;
            }
            byte[] bytes = contents.get(index);
            ByteBuffer buffer = ByteBuffer.wrap(bytes).order(order);
            for (int position = 0; position + pointerSize * 3 <= bytes.length; position += pointerSize) {
                long nameAddress = readPointer(buffer, position, pointerSize);
                List<Row> candidates = byNameAddress.get(nameAddress);
                if (candidates == null) {
                    continue;
                }
                long descriptorAddress = readPointer(buffer, position + pointerSize, pointerSize);
                long functionAddress = readPointer(buffer, position + pointerSize * 2, pointerSize);
                for (Row row : candidates) {
                    if (resolved.contains(row)
                            || !stringAddresses.getOrDefault(row.descriptor(), Set.of()).contains(descriptorAddress)) {
                        continue;
                    }
                    long masked = isThumbCapable() ? functionAddress & ~1L : functionAddress;
                    Address target = toAddr(masked);
                    if (target == null || !memory.contains(target)) {
                        continue;
                    }
                    annotate(target, row, "Registered through RegisterNatives");
                    resolved.add(row);
                    println("JByteMod: " + row.className() + "." + row.method() + row.descriptor() + " -> " + target);
                }
            }
        }
        return resolved.size();
    }

    private boolean isThumbCapable() {
        String processor = currentProgram.getLanguage().getProcessor().toString();
        return processor.equals("ARM");
    }

    private static long readPointer(ByteBuffer buffer, int position, int size) {
        return size == 8 ? buffer.getLong(position) : buffer.getInt(position) & 0xffffffffL;
    }

    private static int indexOf(byte[] data, byte[] pattern, int from) {
        outer:
        for (int index = from; index <= data.length - pattern.length; index++) {
            for (int offset = 0; offset < pattern.length; offset++) {
                if (data[index + offset] != pattern[offset]) {
                    continue outer;
                }
            }
            return index;
        }
        return -1;
    }

    private static List<String> parseArguments(String descriptor) {
        List<String> result = new ArrayList<>();
        int index = descriptor.indexOf('(') + 1;
        int end = descriptor.indexOf(')');
        while (index > 0 && index < end) {
            int dimensions = 0;
            while (descriptor.charAt(index) == '[') {
                dimensions++;
                index++;
            }
            String type;
            char kind = descriptor.charAt(index);
            if (kind == 'L') {
                int semicolon = descriptor.indexOf(';', index);
                type = descriptor.substring(index + 1, semicolon).replace('/', '.');
                index = semicolon + 1;
            } else {
                type = primitive(kind);
                index++;
            }
            result.add(type + "[]".repeat(dimensions));
        }
        return result;
    }

    private static String parseReturn(String descriptor) {
        int close = descriptor.indexOf(')');
        if (close < 0 || close + 1 >= descriptor.length()) {
            return "void";
        }
        String text = descriptor.substring(close + 1);
        int dimensions = 0;
        while (text.startsWith("[")) {
            dimensions++;
            text = text.substring(1);
        }
        String type = text.startsWith("L") ? text.substring(1, text.length() - 1).replace('/', '.') : primitive(text.charAt(0));
        return type + "[]".repeat(dimensions);
    }

    private static String primitive(char kind) {
        return switch (kind) {
            case 'Z' -> "boolean";
            case 'B' -> "byte";
            case 'C' -> "char";
            case 'S' -> "short";
            case 'I' -> "int";
            case 'J' -> "long";
            case 'F' -> "float";
            case 'D' -> "double";
            default -> "void";
        };
    }
}
